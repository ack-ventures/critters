# ACK-213: Add Issue Creation Modal to Dashboard UI

## Summary

Add a "New Critter" button to the dashboard header and a modal form for creating new critter tickets. The form populates team and critter type dropdowns from `/api/v1/metadata` and submits to `/api/v1/issues`. Also add auth token handling for all POST requests from the dashboard using `localStorage` for token persistence when `dashboardToken` is not embedded server-side.

## Files to Modify

#### 1a. Auth helper and auth check script (inline `<script>` at top of script section, ~line 819)

Currently the dashboard embeds the token server-side via `var __dashboardToken = ...` (line 820). For the new modal, we need a `getAuthHeaders()` helper that works both with the server-side token and a `localStorage` fallback. Add the following at the beginning of the `<script>` block (after line 820):

```javascript
function getAuthHeaders() {
  var token = __dashboardToken || localStorage.getItem('critters-token');
  return token ? { 'Authorization': 'Bearer ' + token } : {};
}
```

Then add an auth check IIFE that:
1. Fetches `GET /api/v1/auth-check`
2. If `required: true` and no token available (`!__dashboardToken && !localStorage.getItem('critters-token')`), shows an inline auth bar (a `div#auth-prompt` element that's added to the HTML but hidden by default)
3. The auth bar contains a password input + "Save" button. On submit, stores the token in `localStorage` and hides the bar
4. On any POST returning 401, clears `localStorage` token and shows the auth bar again

Update the existing Poll Now button handler (lines 894-922) to use `getAuthHeaders()` instead of the manual header construction:
```javascript
// Change from:
var headers = {};
if (__dashboardToken) {
  headers['Authorization'] = 'Bearer ' + __dashboardToken;
}
// To:
var headers = getAuthHeaders();
```

#### 1e. Modal JavaScript — inline `<script>` (after existing scripts, before `</script>`)

Add an IIFE that handles all modal logic:

```javascript
(function() {
  var modal = document.getElementById('create-modal');
  var openBtn = document.getElementById('new-critter-btn');
  var cancelBtn = document.getElementById('create-cancel');
  var form = document.getElementById('create-form');
  var providerWrap = document.getElementById('create-provider-wrap');
  var providerSelect = document.getElementById('create-provider');
  var teamSelect = document.getElementById('create-team');
  var typeSelect = document.getElementById('create-type');
  var errorEl = document.getElementById('create-error');
  var successEl = document.getElementById('create-success');
  var submitBtn = document.getElementById('create-submit');
  var metadataCache = null;

  function hideModal() {
    modal.style.display = 'none';
    paused = false; // Reference the outer-scoped paused variable
  }

  function showModal() {
    modal.style.display = 'flex';
    paused = true;
    errorEl.style.display = 'none';
    successEl.style.display = 'none';
    form.reset();
    submitBtn.disabled = false;
    submitBtn.textContent = 'Create';
    loadMetadata();
  }

  openBtn.addEventListener('click', showModal);
  cancelBtn.addEventListener('click', hideModal);
  modal.addEventListener('click', function(e) { if (e.target === modal) hideModal(); });
  document.addEventListener('keydown', function(e) { if (e.key === 'Escape' && modal.style.display !== 'none') hideModal(); });

  function loadMetadata() {
    if (metadataCache) { populateDropdowns(metadataCache); return; }
    fetch('/api/v1/metadata').then(function(r) { return r.json(); }).then(function(data) {
      metadataCache = data;
      populateDropdowns(data);
    }).catch(function() {
      errorEl.textContent = 'Failed to load metadata';
      errorEl.style.display = 'block';
    });
  }

  function populateDropdowns(data) {
    var providers = Object.keys(data.providers || {});
    // Provider select
    providerSelect.innerHTML = '';
    providers.forEach(function(p) {
      var opt = document.createElement('option');
      opt.value = p; opt.textContent = p;
      providerSelect.appendChild(opt);
    });
    providerWrap.style.display = providers.length > 1 ? '' : 'none';

    // Critter types
    typeSelect.innerHTML = '';
    var types = data.critterTypes || [];
    types.forEach(function(ct) {
      var opt = document.createElement('option');
      opt.value = ct.name; opt.textContent = ct.name + ' (' + ct.triggerLabel + ')';
      typeSelect.appendChild(opt);
    });

    updateTeams(data);
  }

  // Bind provider change listener once (outside populateDropdowns to avoid duplication across refresh cycles)
  providerSelect.addEventListener('change', function() {
    if (metadataCache) updateTeams(metadataCache);
  });

  function updateTeams(data) {
    var provider = providerSelect.value;
    var teams = (data.providers[provider] || {}).teams || [];
    teamSelect.innerHTML = '';
    teams.forEach(function(t) {
      var opt = document.createElement('option');
      opt.value = t.id; opt.textContent = t.name + ' (' + t.key + ')';
      teamSelect.appendChild(opt);
    });
  }

  function escapeText(s) {
    var d = document.createElement('div');
    d.textContent = s;
    return d.innerHTML;
  }

  form.addEventListener('submit', function(e) {
    e.preventDefault();
    errorEl.style.display = 'none';
    successEl.style.display = 'none';
    submitBtn.disabled = true;
    submitBtn.textContent = 'Creating...';

    var body = {
      provider: providerSelect.value,
      teamId: teamSelect.value,
      title: document.getElementById('create-title').value,
      description: document.getElementById('create-description').value,
      critterType: typeSelect.value
    };

    fetch('/api/v1/issues', {
      method: 'POST',
      headers: Object.assign({ 'Content-Type': 'application/json' }, getAuthHeaders()),
      body: JSON.stringify(body)
    })
    .then(function(res) {
      if (res.status === 401) {
        localStorage.removeItem('critters-token');
        showAuthPrompt();
        throw new Error('Unauthorized - please set your token');
      }
      return res.json();
    })
    .then(function(data) {
      if (data.success) {
        var msg = 'Created ' + escapeText(data.identifier);
        if (data.url) {
          var a = document.createElement('a');
          a.href = data.url;
          a.target = '_blank';
          a.rel = 'noopener';
          a.textContent = 'View';
          msg += ' \u2014 ';
          successEl.innerHTML = msg;
          successEl.appendChild(a);
        } else {
          successEl.textContent = msg;
        }
        successEl.style.display = 'block';
        // Trigger a poll so the new issue gets picked up
        fetch('/poll', { method: 'POST', headers: getAuthHeaders() }).catch(function() {});
        setTimeout(hideModal, 5000);
      } else {
        errorEl.textContent = data.error || 'Unknown error';
        errorEl.style.display = 'block';
        submitBtn.disabled = false;
        submitBtn.textContent = 'Create';
      }
    })
    .catch(function(err) {
      errorEl.textContent = err.message || 'Request failed';
      errorEl.style.display = 'block';
      submitBtn.disabled = false;
      submitBtn.textContent = 'Create';
    });
  });
})();
```

To allow the modal IIFE to set `paused = true`, we need to hoist `paused` to the outer script scope. Change line 929 from `var paused = false;` to instead declare `paused` at the top of the script block (right after `var __dashboardToken = ...;`):

```javascript

var paused = false;
```

And remove the `var paused = false;` from line 929 inside the refresh IIFE (it will now use the outer-scoped variable).

#### 1f. Auth check IIFE and showAuthPrompt function

Add in the script section (after `getAuthHeaders` definition):

```javascript
function showAuthPrompt() {
  var prompt = document.getElementById('auth-prompt');
  if (prompt) prompt.style.display = 'block';
}

(function() {
  var saveBtn = document.getElementById('auth-save-btn');
  var tokenInput = document.getElementById('auth-token-input');
  if (!saveBtn || !tokenInput) return;

  saveBtn.addEventListener('click', function() {
    var val = tokenInput.value.trim();
    if (val) {
      localStorage.setItem('critters-token', val);
      document.getElementById('auth-prompt').style.display = 'none';
    }
  });

  // On load, check if auth is required
  fetch('/api/v1/auth-check').then(function(r) { return r.json(); }).then(function(data) {
    if (data.required && !__dashboardToken && !localStorage.getItem('critters-token')) {
      showAuthPrompt();
    }
  }).catch(function() {});
})();
```

## Edge Cases and Considerations

1. **Single provider**: The provider dropdown is hidden when only one provider exists (most common case)
2. **No critter types**: If `critterTypes` is empty in metadata, the type dropdown will be empty — the API still works since it defaults the label to "Critter"
3. **No teams**: If a provider returns no teams, the team dropdown is empty and the required `teamId` validation on the server will reject the submission with a clear error
4. **Auth flow**: Three auth scenarios are handled:
   - `dashboardToken` embedded server-side (current behavior, no change needed)
   - No token configured (auth-check returns `required: false`, no prompt shown)
   - Token required but not embedded (e.g., remote access via tunnel) — user sees auth prompt, token stored in `localStorage`
5. **Auto-refresh pause**: The `paused` variable is set to `true` when modal opens and `false` when it closes, preventing mid-form page refreshes
6. **XSS defense-in-depth in success message**: The success message uses `escapeText()` for `data.identifier` and constructs the View link via `document.createElement('a')` with `setAttribute` to avoid innerHTML injection, even though the tracker API is trusted internal infrastructure
7. **Modal close on success**: Auto-closes after 5 seconds via `setTimeout(hideModal, 5000)`, giving users enough time to see the identifier and click the View link
8. **Poll trigger after creation**: After successful creation, the modal fires a background `POST /poll` to trigger immediate pickup of the new issue
9. **`providerSelect` change listener**: The change listener is bound once in the outer IIFE scope (outside `populateDropdowns`), using `metadataCache` to get the data. This avoids listener duplication even across `doRefresh` cycles that re-execute all scripts, since each script re-execution creates a fresh IIFE with its own single listener binding

## Previous Review Items

> [SHOULD_FIX] The `providerSelect` change listener duplication analysis in edge case 9 is based on incorrect reasoning. The plan states "metadataCache ensures populateDropdowns is only called once" but `metadataCache` is local to each IIFE invocation and resets to `null` after every `doRefresh` cycle. The actual mitigation is that `paused = true` prevents refresh while the modal is open, so duplication accumulates slowly (once per open/close cycle). To properly fix this, the listener should be bound outside `populateDropdowns` using a flag variable like `var listenerBound = false` that is checked before adding the listener.

Addressed: Moved the `providerSelect.addEventListener('change', ...)` call outside of `populateDropdowns` and into the outer IIFE scope. The listener now uses `metadataCache` (which is set by `loadMetadata`) to get data when the provider changes. This means only one listener is bound per IIFE instantiation, and since `doRefresh` replaces the DOM elements entirely (including `providerSelect`), the old listener is garbage-collected with the old element. Each fresh script execution binds exactly one listener to the new element. No duplication occurs.

> [SHOULD_FIX] The auth-prompt HTML (section 1b) contains a redundant duplicate `display:none` in the inline style: the style starts with `display:none;` and also ends with `display:none;`. Remove the duplicate occurrence for clarity.

Addressed: Removed the duplicate `display:none` from the auth-prompt HTML. The style now contains only a single `display:none` at the beginning.

> [CONSIDER] The success message in the modal uses `successEl.innerHTML = msg` where `msg` includes `data.url` from the tracker API. While the data source is trusted internal infrastructure, a simple HTML-entity escape of `data.identifier` and `data.url` before injection would be a low-cost defense-in-depth measure.
