# ACK-246: Dashboard — Add Filterable Recent Activity Table

## Summary

Add client-side filtering to the recent activity table on the Critters dashboard. Users will be able to filter by **critter type** and **status** using clickable badge-style buttons rendered above the table. Clicking a badge in a table row also activates that filter. Filter state persists in URL query params so it survives the 30-second auto-refresh. A row counter shows "Showing X of Y" and a "Clear filters" button appears when filters are active.

## Files to Modify

### `src/dashboard.ts` (sole file)

All changes are in this single file since the dashboard is a self-contained server-rendered HTML page with inline CSS and JS.

```typescript
const activityStatuses = [...new Set(recentActivity.map(m => {

  const isOk = m.event === "task_completed" || m.event === "review_completed";
```

### 3. Add CSS for filter button states


### 4. Add filter JavaScript (new IIFE script block)

**Location:** After the existing text-filter IIFE (line 986), add a new `<script>` block with a new IIFE.

```javascript
(function() {
  var filterBar = document.getElementById('activity-filters');
  var table = document.querySelector('.table-section table');
  if (!filterBar || !table) return;
  var tbody = table.querySelector('tbody');
  var counterEl = document.getElementById('row-counter');
  var clearBtn = document.getElementById('clear-filters-btn');
  var textInput = document.getElementById('activity-filter');

  var activeFilters = { type: '', status: '' };

  // Read initial filter state from URL
  var params = new URLSearchParams(window.location.search);
  var initType = params.get('ftype') || '';
  var initStatus = params.get('fstatus') || '';
  activeFilters.type = initType;
  activeFilters.status = initStatus;

  // Highlight initial active buttons
  filterBar.querySelectorAll('[data-filter-group]').forEach(function(btn) {
    var group = btn.getAttribute('data-filter-group');
    var val = btn.getAttribute('data-filter-value');
    btn.classList.toggle('active', val === activeFilters[group]);
  });

  function updateURL() {
    var url = new URL(window.location);
    if (activeFilters.type) url.searchParams.set('ftype', activeFilters.type);
    else url.searchParams.delete('ftype');
    if (activeFilters.status) url.searchParams.set('fstatus', activeFilters.status);
    else url.searchParams.delete('fstatus');
    history.replaceState(null, '', url.toString());
  }

  function applyFilters() {
    var rows = tbody.querySelectorAll('tr');
    var total = rows.length;
    var visible = 0;
    var textQuery = textInput ? textInput.value.toLowerCase() : '';

    rows.forEach(function(row) {
      var matchType = !activeFilters.type || row.getAttribute('data-type') === activeFilters.type;
      var matchStatus = !activeFilters.status || row.getAttribute('data-status') === activeFilters.status;
      var matchText = !textQuery || row.textContent.toLowerCase().includes(textQuery);
      var show = matchType && matchStatus && matchText;
      row.style.display = show ? '' : 'none';
      if (show) visible++;
    });

    // Update counter
    if (counterEl) {
      var hasFilter = activeFilters.type || activeFilters.status || textQuery;
      counterEl.textContent = hasFilter ? 'Showing ' + visible + ' of ' + total : total + ' entries';
    }

    // Show/hide clear button
    if (clearBtn) {
      clearBtn.style.display = (activeFilters.type || activeFilters.status) ? '' : 'none';
    }

    updateURL();
  }

  // Filter button clicks
  filterBar.addEventListener('click', function(e) {
    var btn = e.target.closest('[data-filter-group]');
    if (!btn) {
      // Clear button
      if (e.target.id === 'clear-filters-btn' || e.target.closest('#clear-filters-btn')) {
        activeFilters.type = '';
        activeFilters.status = '';
        filterBar.querySelectorAll('[data-filter-group]').forEach(function(b) {
          b.classList.toggle('active', b.getAttribute('data-filter-value') === '');
        });
        applyFilters();
      }
      return;
    }

    var group = btn.getAttribute('data-filter-group');
    var val = btn.getAttribute('data-filter-value');
    activeFilters[group] = val;

    // Update active state for this group
    filterBar.querySelectorAll('[data-filter-group="' + group + '"]').forEach(function(b) {
      b.classList.toggle('active', b.getAttribute('data-filter-value') === val);
    });

    applyFilters();
  });

  // Make table badges clickable
  tbody.addEventListener('click', function(e) {
    var badge = e.target.closest('.badge');
    if (!badge) return;
    var row = badge.closest('tr');
    if (!row) return;

    // Determine if this is a type or status badge based on cell index
    var cell = badge.closest('td');
    var cellIdx = Array.from(row.cells).indexOf(cell);

    if (cellIdx === 1) {
      // Type column — but type isn't in a badge, it's plain text. Skip.
      return;
    }
    if (cellIdx === 2) {
      // Status column
      var statusVal = row.getAttribute('data-status');
      activeFilters.status = statusVal;
      filterBar.querySelectorAll('[data-filter-group="status"]').forEach(function(b) {
        b.classList.toggle('active', b.getAttribute('data-filter-value') === statusVal);
      });
      applyFilters();
    }
  });

  // Also make type column text clickable
  tbody.addEventListener('click', function(e) {
    var cell = e.target.closest('td');
    if (!cell) return;
    var row = cell.closest('tr');
    if (!row) return;
    var cellIdx = Array.from(row.cells).indexOf(cell);
    if (cellIdx !== 1) return; // Only type column (index 1)
    // Don't interfere with links
    if (e.target.tagName === 'A') return;

    var typeVal = row.getAttribute('data-type');
    activeFilters.type = typeVal;
    filterBar.querySelectorAll('[data-filter-group="type"]').forEach(function(b) {
      b.classList.toggle('active', b.getAttribute('data-filter-value') === typeVal);
    });
    applyFilters();
  });

  // Override the existing text filter to work with badge filters
  if (textInput) {
    // Remove old listener by replacing the element
    var newInput = textInput.cloneNode(true);
    textInput.parentNode.replaceChild(newInput, textInput);
    newInput.addEventListener('input', function() { applyFilters(); });
    textInput = newInput;
  }

  // Apply initial filters
  applyFilters();
})();
```

### 5. Update auto-refresh to preserve filter state

**Location:** The refresh IIFE (lines 1015–1067)

The `doRefresh()` function already uses `window.location.search` when fetching the refreshed HTML (line 1034):
```javascript
var url = '/dashboard' + window.location.search;
```

Since we persist `ftype` and `fstatus` in URL query params via `history.replaceState`, the auto-refresh will automatically include these params in the fetch URL. The server renders the same HTML regardless (the `ftype`/`fstatus` params are client-side only), and after the HTML is replaced and scripts re-execute, the filter IIFE reads from `window.location.search` to restore state.

**No server-side changes needed** — the filter params are purely client-side.

The existing `paused` logic on `filterInput.focus/blur` (lines 1022–1025) should also be extended to pause refresh when any filter button is being interacted with. However, since button clicks are instantaneous (not sustained like text input focus), this is unnecessary — the 30s refresh won't interrupt a click.

## URL Query Param Design

- `ftype` — active type filter value (e.g., `?ftype=create`)
- `fstatus` — active status filter value (e.g., `?fstatus=Failed`)
- Both are optional; absence means "all"
- These coexist with the existing `type` param used by the top-level type filter buttons (which is a server-side filter applied to all dashboard sections, not just the table)
- Use `history.replaceState` (not `pushState`) to avoid polluting browser history

## Edge Cases

1. **Empty table** — if `recentActivity` is empty, the filter bar is still rendered but with no type/status buttons (just "All" buttons). The counter shows "0 entries".
2. **Single type/status** — filter bar still shows to maintain layout consistency, but the single filter button is visually obvious.
3. **Text filter + badge filters** — both apply simultaneously (AND logic). The row counter reflects the combined result.
4. **Auto-refresh resets DOM** — after refresh, all scripts re-execute. The filter IIFE reads from `window.location.search` to restore state, so filters survive refresh.
5. **Existing top-level type filter** — the server-side `?type=X` filter already reduces `recentActivity` to one type. The table-level type filter badges will then show only that single type. This is correct and not confusing.
6. **Sort + filter** — sorting operates on visible rows since `tbody.querySelectorAll('tr')` includes hidden ones. However, `appendChild` just reorders all rows (hidden ones too), which is fine — hidden rows remain hidden after re-sort.
